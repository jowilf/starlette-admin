$(document).on("click", ".copy-to-clipboard", function () {
  var btn = $(this);
  var icon = btn.find("i");
  navigator.clipboard.writeText(btn.attr("data-copy-value")).then(function () {
    icon.removeClass("fa-copy").addClass("fa-check");
    setTimeout(function () {
      icon.removeClass("fa-check").addClass("fa-copy");
    }, 1500);
  });
});
